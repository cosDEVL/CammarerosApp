class AppError extends Error {
  constructor(statusCode, message, errors = null) {
    super(message);

    this.statusCode = statusCode;
    this.status = `${statusCode}`.startsWith(`4`) ? `Fail` : `Error`;
    this.errors = errors;
    this.isOperational = true;

    Error.captureStackTrace(this, this.constructor);
  }
}

module.exports = AppError;

const { findValidRefreshToken } = require("../repositories/token.repository");
const AppError = require("../utils/appError");
const { clearAuthCookie } = require("../utils/cookie");
const { verifyRefreshToken, hashToken } = require("../utils/token");

module.exports = async (req, res, next) => {

  const refreshToken = req.cookies?.refresh_token;
  if (!refreshToken) return next(new AppError(401, "Refresh Token missing"))

  try {
    verifyRefreshToken(refreshToken);

    const hashRefresh = hashToken(refreshToken);
    const session = await findValidRefreshToken(hashRefresh)

    if (!session) throw new Error("Refresh token revoked or user inactive")

    req.user = { id: session.user_id, role: session.role }
    req.refreshTokenHash = hashRefresh;
    next();

  } catch (err) {
    clearAuthCookie(res);
    return next(new AppError(401, "Session invalid or expired. Please Log-In again"))
  }
}

const sql = require('../config/postgresDB');

const insertHashToken = async (user_id, hashToken) => {

  await sql`
        INSERT INTO refresh_tokens (token_hash, user_id, expires_at)
        VALUES (${hashToken}, ${user_id}, NOW() + INTERVAL '30 days');
    `;
};

const rotateHashToken = async (oldRefreshToken, newRefreshToken) => {

  await sql`
        UPDATE refresh_tokens
        SET token_hash = ${newRefreshToken},
            expires_at = NOW() + INTERVAL '30 days'
        WHERE token_hash = ${oldRefreshToken}
    `;

}

const findValidRefreshToken = async (tokenHash) => {

  const [session] = await sql`
        SELECT rt.token_hash, u.id as user_id, u.role
        FROM refresh_tokens rt JOIN users u ON rt.user_id = u.id
        WHERE rt.token_hash = ${tokenHash}
        AND rt.expires_at > NOW()
        AND u.is_active = true
    `;


  return session;
};

const deleteHashToken = async (tokenHash) => {
  await sql`
        DELETE FROM refresh_tokens
        WHERE token_hash = ${tokenHash}
    `
};

module.exports = { insertHashToken, findValidRefreshToken, deleteHashToken, rotateHashToken };

const isProd = process.env.NODE_ENV === 'production';

const baseCookieOptions = {
  httpOnly: true,
  secure: isProd,
  sameSite: 'lax'
};

const accessCookieOptions = {
  ...baseCookieOptions,
  path: '/'
};

const refreshCookieOptions = {
  ...baseCookieOptions,
  path: '/api/v1/auth'
};

const setAuthCookie = (res, { accessToken, refreshToken }) => {
  res.cookie('access_token', accessToken, {
    ...accessCookieOptions,
    maxAge: 15 * 60 * 1000 //15min
    // maxAge: 10000 //10s
  });

  res.cookie('refresh_token', refreshToken, {
    ...refreshCookieOptions,
    maxAge: 30 * 24 * 60 * 60 * 1000 //30days
  })
};

const clearAuthCookie = (res) => {
  res.clearCookie('access_token', accessCookieOptions);
  res.clearCookie('refresh_token', refreshCookieOptions);
}

module.exports = { setAuthCookie, clearAuthCookie }

const jwt = require('jsonwebtoken');
const crypto = require('crypto');

const signAccessToken = (user) => {
  return jwt.sign(
    { id: user.id, role: user.role },
    process.env.JWT_ACCESS_SECRET,
    { expiresIn: '15m' }
  );
};

const verifyAccessToken = (token) => {
  return jwt.verify(token, process.env.JWT_ACCESS_SECRET);
};

const signRefreshToken = (user) => {
  return jwt.sign(
    { id: user.id, jti: crypto.randomUUID() },
    process.env.JWT_REFRESH_SECRET,
    { expiresIn: '30d' }
  )
}

const verifyRefreshToken = (token) => {
  return jwt.verify(token, process.env.JWT_REFRESH_SECRET)
}

const hashToken = (token) => {
  return crypto.createHash('sha256').update(token).digest('hex');
}

module.exports = {
  signAccessToken,
  verifyAccessToken,
  signRefreshToken,
  verifyRefreshToken,
  hashToken
}

  await setCookies(user, res);

  res.status(200).json({
    status: 'ok',
    message: 'User logged in successfully!',
    data: { user }
  })

const logoutUser = async (req, res, next) => {

  await clearCookies(req.cookies.refresh_token, res)

  res.status(200).json({
    status: 'ok',
    message: 'User logged out successfully!'
  })
}

const refreshToken = async (req, res, next) => {

  await rotateCookie(req.user, req.refreshTokenHash, res);

  res.status(200).json({
      status: 'ok',
      message: 'Token refreshed successfully',
      user: { user: req.user },
    });
}

module.exports = { registerUser, loginUser, logoutUser, refreshToken };

const bcrypt = require('bcryptjs');
const AppError = require('../utils/appError');
const authRepo = require('../repositories/auth.repository');

const registerUser = async ({ name, lastname, phone, email, password, invite_code }) => {

  const inviteCode = await authRepo.getInviteCode();

  if (invite_code !== inviteCode)
    throw new AppError(400, "Enter the correct invite code");

  const passwordHash = await bcrypt.hash(password, 10);

  const user = await authRepo.createUser({ name, lastname, phone, email, passwordHash })

  return user;
}

const loginUser = async ({ email, password }) => {

  const user = await authRepo.getUser(email);

  if (
    !user ||
    !(await bcrypt.compare(password, user.password_hash)) ||
    !user.is_active
  ) throw new AppError(400, "Email or Password incorrect");

  const { id, name, lastname, phone, role } = user;
  return { id, name, lastname, phone, email: user.email, role };
}


module.exports = { registerUser, loginUser };

})

  password: z.string().trim().min(1, 'Password required')
})
